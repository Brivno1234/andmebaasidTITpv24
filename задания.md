CREATE TRIGGER KyberSportKustutamine
ON KyberSport
FOR DELETE
AS
INSERT INTO logi(kasutaja, kuupaev, sisestatudAndmed)
SELECT
SYSTEM_USER,
GETDATE(),
CONCAT('KustutadKyberSport: ', deleted.KyberRyhmNimi, ', Mang: ', deleted.MangID)
FROM deleted

DELETE FROM KyberSport WHERE PKKyberSportID=4;

@nimi VARCHAR(50)

BEGIN

@vanus INT
AS
BEGIN

    SELECT *
    FROM KyberOsaleja
    WHERE Vanus <= @vanus;

END;

--view

CREATE VIEW SportMang
AS
SELECT
	KyberSport.PKKyberSportID,
    KyberSport.KyberRyhmNimi,
    KyberSport.OsalejateArv,
    Mang.MangNimi
FROM KyberSport INNER JOIN Mang ON KyberSport.MangID = Mang.MangID;


ALTER VIEW SuuredTiimid
AS
SELECT
    KyberSport.KyberRyhmNimi,
    KyberSport.OsalejateArv,
    Mang.MangNimi
FROM KyberSport INNER JOIN Mang ON KyberSport.MangID = Mang.MangID WHERE KyberSport.OsalejateArv >= 6;

CREATE VIEW ViewTiimid

AS
SELECT
    KyberSport.KyberRyhmNimi,
    KyberSport.OsalejateArv,
    Mang.MangNimi
FROM KyberSport INNER JOIN Mang ON KyberSport.MangID = Mang.MangID WHERE Mang.MangNimi = 'CS2';


SELECT * FROM SportMang;
